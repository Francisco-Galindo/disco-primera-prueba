  import csv
  import os

  def getMean(n, DC, T, alpha):
      mean = 0
      try:
          with open(f'{os.getcwd()}/discovery-data/{n}-nodes-{round(DC/ 100, 2)}-dc-{T}-T-{alpha}-alpha-first-node-latency.tsv', 'r') as file:
              tsv_reader = csv.reader(file, delimiter='\t')
              ac = 0
              population = 0
              for row in tsv_reader:
                  try:
                      lat = int(row[0])
                      freq = int(row[1])

                      term = lat * freq

                      population += freq
                      ac += term
                  except:
                      continue

              mean = ac / population

          return mean
      except:
          return float('nan')

  def harmonic(n):
      s = 0
      for i in range(1, n+1):
          s += 1 / i

      return s

  T = 100
  DC = 5
  alpha = 0.125
  L = 1
  for n in [2, 5, 10, 20, 50]:
      numerator = T * harmonic(n-1)
      p_succ_1 = ((2 * DC**2 * (L - 3 * alpha)) / T)  *  (1 - (4 * DC * alpha)/T)**(n-2)
      p_succ_2 = (2 * DC**2 * (L - 3 * alpha) / T) * (1 - (2*alpha)/T)**(2*DC*(n-2))
      val_inf =  numerator / p_succ_1
      val_sup =  numerator / p_succ_2
      row = [n, round(val_inf), round(val_sup), round(getMean(n, DC, T, alpha))]
      data.append(row)


  return data
